# EPIC 1 — SOCIAL LAYER

## HANDOFF PACKAGE HO-A: Social Graph & Activity Feed
### For: OpenCode → execute (then code-review → audit)
### Golden Template: `037_organizer_follows.sql` (the join table + RLS shape) for schema; `reviews/route.ts` for endpoints; `(public)/my-events/page.tsx` for the feed list page.

### 1. Objective
Add a **user↔user social graph** (distinct from organizer-follows) and a **personalized activity
feed** so attendees see what people they follow are doing ("Sara saved *Jazz Night*", "Dawit is
going to *DevFest*") and see "friends attending" on event pages. This is the core retention moat.

### 2. Architecture Decisions (LOCKED)
- `user_follows(follower_id, following_id)` owner-scoped join, `UNIQUE`, idempotent POST → P1/P2.
- Feed is **materialized**, not computed on read: writes to owned tables (save, register, review,
  follow) append a row to `feed_activities` via **DB triggers** so the feed is a cheap indexed
  read. Fan-out-on-read at query time (join `feed_activities` where actor ∈ people-I-follow) →
  P1 + performance-patterns (avoid N+1 on hot path).
- Activity visibility respects a per-user `profiles.activity_private` flag (default false) — a
  private user's actions never enter followers' feeds → security/privacy.
- Follow is public-readable social proof (follower/following counts, "friends attending"); the
  **feed read** is strictly the caller's own (rows where I follow the actor) → RLS split.

### 3. Schema Specification

-- Migration 038: user-to-user social graph + activity feed
ALTER TABLE public.profiles
  ADD COLUMN IF NOT EXISTS activity_private BOOLEAN NOT NULL DEFAULT false;

CREATE TABLE public.user_follows (
  id            UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  follower_id   UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  following_id  UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (follower_id, following_id),
  CHECK (follower_id <> following_id)
);
CREATE INDEX idx_user_follows_follower  ON public.user_follows(follower_id, created_at DESC);
CREATE INDEX idx_user_follows_following ON public.user_follows(following_id);

ALTER TABLE public.user_follows ENABLE ROW LEVEL SECURITY;
-- Public read = social proof (counts, "who follows whom"); writes owner-only.
CREATE POLICY "user_follows_select_all"  ON public.user_follows FOR SELECT USING (true);
CREATE POLICY "user_follows_insert_own"  ON public.user_follows FOR INSERT WITH CHECK (auth.uid() = follower_id);
CREATE POLICY "user_follows_delete_own"  ON public.user_follows FOR DELETE USING (auth.uid() = follower_id);
GRANT SELECT ON public.user_follows TO anon, authenticated;
GRANT INSERT, DELETE ON public.user_follows TO authenticated;

-- Activity feed (materialized via triggers)
CREATE TYPE public.feed_verb AS ENUM ('saved_event','registered_event','reviewed_event','followed_user','followed_organizer');

CREATE TABLE public.feed_activities (
  id            UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  actor_id      UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  verb          public.feed_verb NOT NULL,
  event_id      UUID REFERENCES public.events(id) ON DELETE CASCADE,
  target_user_id     UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  target_organizer_id UUID REFERENCES public.organizers(id) ON DELETE CASCADE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_feed_actor_time ON public.feed_activities(actor_id, created_at DESC);
CREATE INDEX idx_feed_time       ON public.feed_activities(created_at DESC);

ALTER TABLE public.feed_activities ENABLE ROW LEVEL SECURITY;
-- I can read an activity iff I follow the actor AND the actor is not private, OR I am the actor.
CREATE POLICY "feed_select_followed" ON public.feed_activities FOR SELECT USING (
  actor_id = auth.uid()
  OR EXISTS (
    SELECT 1 FROM public.user_follows uf
    JOIN public.profiles p ON p.id = feed_activities.actor_id
    WHERE uf.follower_id = auth.uid() AND uf.following_id = feed_activities.actor_id
      AND p.activity_private = false
  )
);
-- No direct client writes: rows are inserted by SECURITY DEFINER triggers only.
GRANT SELECT ON public.feed_activities TO authenticated;

-- Trigger fn: append feed rows on owned-table writes. SECURITY DEFINER to bypass RLS on insert.
CREATE OR REPLACE FUNCTION public.fn_feed_on_save() RETURNS trigger
  LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  INSERT INTO public.feed_activities(actor_id, verb, event_id)
  VALUES (NEW.profile_id, 'saved_event', NEW.event_id);
  RETURN NEW;
END $$;
CREATE TRIGGER trg_feed_on_save AFTER INSERT ON public.saved_events
  FOR EACH ROW EXECUTE FUNCTION public.fn_feed_on_save();

CREATE OR REPLACE FUNCTION public.fn_feed_on_register() RETURNS trigger
  LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  IF NEW.status = 'confirmed' THEN
    INSERT INTO public.feed_activities(actor_id, verb, event_id)
    VALUES (NEW.profile_id, 'registered_event', NEW.event_id);
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER trg_feed_on_register AFTER INSERT ON public.registrations
  FOR EACH ROW EXECUTE FUNCTION public.fn_feed_on_register();
-- (Add analogous fn_feed_on_review on reviews, fn_feed_on_follow_user on user_follows,
--  fn_feed_on_follow_org on organizer_follows — same SECURITY DEFINER shape.)
```

  is wanted.

## HANDOFF PACKAGE HO-B: Event Q&A / Discussion
### For: OpenCode → execute (then code-review → audit)
### Golden Template: `010_reviews_ratings.sql` (content table + updated_at trigger + RLS); `reviews/route.ts` + `public/events/[slug]/reviews/route.ts` (public-read + owner-write).

### 1. Objective
Threaded **questions & answers** on event pages: attendees ask, organizers (and attendees) answer,
votes rank answers, organizer can pin. Pre-event engagement that also deflects repetitive DMs.

### 2. Architecture Decisions (LOCKED)
- Two tables: `event_questions` (top-level) and `event_answers` (replies), both owner-writable,
  public-readable → P1/P3. Votes in `event_question_votes` (owner-scoped, idempotent) drive
  answer/question ranking → P2.
- Organizer of the event OR admin may **pin** and **moderate** (soft-hide) any Q/A via a policy
  that checks `organizers.owner`/team membership → security. Attendees edit/delete only their own.
- Reuse the existing `content_moderation` pipeline hook for new questions (queue for AI moderation
  cron) — consume, do not modify the AI seam.

    SELECT 1 FROM public.events e

    LEFT JOIN public.organizer_team_members tm ON tm.organizer_id = o.id AND tm.profile_id = p_user

CREATE POLICY "eq_insert_own" ON public.event_questions FOR INSERT WITH CHECK (auth.uid() = author_id);

CREATE POLICY "eq_delete_own_or_host" ON public.event_questions FOR DELETE

## HANDOFF PACKAGE HO-C: Collections & Curated Lists
### For: OpenCode → execute (then code-review → audit)
### Golden Template: `010_reviews_ratings.sql` (owner content) + `036_saved_events.sql` (join item table); `(public)/categories/[slug]/page.tsx` for the grid page.

### 1. Objective
**Collections** = named, ordered groups of events. Two flavors from one schema: **editorial**
(admin-owned, e.g. "This Weekend in Addis") featured on discovery, and **user "My Lists"**
(public or private, shareable). Luma/Fever-style curation.

### 2. Architecture Decisions (LOCKED)
- One `collections` table with `owner_id` (nullable for editorial → owned by admin actor),
  `is_editorial`, `visibility ('public'|'private'|'unlisted')`, `slug` for shareable URLs → P1.
- `collection_items` join (collection_id, event_id, position) `UNIQUE` → ordered, idempotent add → P2.
- Only admins set `is_editorial`/feature; users CRUD their own; public/unlisted are readable by
  slug, private only by owner → RLS.

### 3. Schema Specification

  EXISTS (SELECT 1 FROM public.collections c WHERE c.id = collection_id AND c.owner_id = auth.uid()));

# EPIC 2 — ENGAGEMENT & RETENTION

## HANDOFF PACKAGE HO-D: Gamification (Badges · Points · Streaks)
### For: OpenCode → execute (then code-review → audit)
### Golden Template: `010_reviews_ratings.sql` for tables; award logic follows the `035_waitlist_auto_promote` server-function pattern (DB function invoked from a route/trigger).

### 1. Objective
Reward attendance and contribution: **badges** (first event, 5-event streak, explorer/5-categories,
super-host), a **points ledger**, and **attendance streaks**, surfaced as a profile trophy case.
Drives repeat behavior.

### 2. Architecture Decisions (LOCKED)
- `badges` = catalog (seeded, admin-managed). `user_badges` = awards (owner-readable, system-written).

## HANDOFF PACKAGE HO-G: Ticket Transfers & Resale
### For: OpenCode → execute (then code-review → audit)
### Golden Template: `007_tickets_registrations.sql` + `022_cancel_registration_and_ticket_uniqueness.sql` (ticket lifecycle + QR uniqueness); `019_atomic_registration.sql` (atomic SECURITY DEFINER fn) for the transfer transaction.

### 1. Objective
Let a ticket holder **transfer** a ticket to another user by email, and optionally **resell** at
face value to the event's waitlist. Transfer atomically reassigns ownership and **invalidates the
old QR / issues a new HMAC code** so the original can't be reused.

### 2. Architecture Decisions (LOCKED)
- `ticket_transfers` logs every transfer (from→to, reason, status) for audit → P1/security.
- The transfer itself is a single `SECURITY DEFINER` fn `fn_transfer_ticket` that, in one
  transaction: verifies caller owns the ticket + event hasn't started + ticket is valid, reassigns
  `tickets.profile_id`/registration, **rotates the QR/HMAC payload**, and writes the transfer row →
  atomicity + HMAC stays server-side (never trust client).
- Resale to waitlist reuses the existing waitlist promotion (`035`) — a resold ticket offers the
  seat to the next waitlisted user; **no peer-to-peer pricing/payments** in V2.0 (face value only,
  payment seam stubbed) → boundary.

CREATE OR REPLACE FUNCTION public.fn_transfer_ticket(p_ticket UUID, p_to_email TEXT, p_kind TEXT)
  RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE v_owner UUID; v_event UUID; v_start TIMESTAMPTZ; v_to UUID; v_transfer UUID;
BEGIN
  SELECT t.profile_id, r.event_id INTO v_owner, v_event
    FROM public.tickets t JOIN public.registrations r ON r.id = t.registration_id
    WHERE t.id = p_ticket FOR UPDATE;
  IF v_owner IS NULL OR v_owner <> auth.uid() THEN RAISE EXCEPTION 'not_owner'; END IF;
  SELECT start_date INTO v_start FROM public.events WHERE id = v_event;
  IF v_start <= now() THEN RAISE EXCEPTION 'event_started'; END IF;
  SELECT id INTO v_to FROM public.profiles WHERE lower(email) = lower(p_to_email);
  INSERT INTO public.ticket_transfers(ticket_id, from_profile, to_profile, to_email, kind,
    status, completed_at)
    VALUES (p_ticket, auth.uid(), v_to, p_to_email, p_kind,
      CASE WHEN v_to IS NULL THEN 'pending' ELSE 'completed' END,
      CASE WHEN v_to IS NULL THEN NULL ELSE now() END)
    RETURNING id INTO v_transfer;
  IF v_to IS NOT NULL THEN
    -- reassign + rotate QR/HMAC payload (regenerate qr_data so old code is dead)
    UPDATE public.tickets
      SET profile_id = v_to, qr_data = encode(gen_random_bytes(24),'hex'), updated_at = now()
      WHERE id = p_ticket;
    -- reassign registration ownership as well
    UPDATE public.registrations SET profile_id = v_to
      WHERE id = (SELECT registration_id FROM public.tickets WHERE id = p_ticket);
  END IF;
  RETURN v_transfer;
END $$;
```

> or on an explicit accept route.
